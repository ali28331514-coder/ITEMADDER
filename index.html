// استبدل دالة fixZipTimestamps بهذا الكود الجديد: fixBtn.addEventListener('click', async () => { console.log('🔧 Starting ZIP fix...', zipState.uploadedFile.name); if (!zipState.uploadedFile) return; fixBtn.disabled = true; fixBtn.innerHTML = 'PROCESSING...'; showZipProgress(true); hideEl('zipErrorBadge'); hideEl('zipSuccessBadge'); hideEl('zipStatsEl'); zipDownloadBtn.style.display = 'none'; try { // 🎯 الحل: استخدام checkCRC + lazy loading setZipProgress(5, 'Validating ZIP...'); await sleep(200); // جرب أولاً الطريقة الآمنة let srcZip; try { srcZip = await JSZip.loadAsync(zipState.uploadedFile, { checkCRC32: true, createFolders: true }); } catch (e1) { console.log('⚠️ Strict mode failed, trying relaxed...'); // الطريقة المرنة للـ ZIP المعطلة srcZip = await JSZip.loadAsync(zipState.uploadedFile, { checkCRC32: false, createFolders: true, decodeFileName: false }); } const entries = Object.keys(srcZip.files); console.log(`📁 Found ${entries.length} files:`, entries.slice(0, 5)); setZipProgress(15, `Scanning ${entries.length} files...`); await sleep(300); const outZip = new JSZip(); const now = new Date(); let fixed = 0; let skipped = 0; let errors = 0; const BAD_YEARS = [1899, 1900, 1970]; // معالجة كل ملف بأمان for (let i = 0; i < entries.length; i++) { const path = entries[i]; const entry = srcZip.files[path]; // Progress update if (i % 20 === 0 || i === entries.length - 1) { const pct = 15 + Math.floor((i / entries.length) * 70); setZipProgress(pct, `Processing ${i+1}/${entries.length} files...`); await sleep(5); } try { if (entry.dir) { outZip.folder(path); continue; } // فحص التاريخ const date = entry.date; const year = date ? date.getFullYear() : 0; const isBad = !date || BAD_YEARS.includes(year) || year < 1980 || isNaN(year); if (isBad) fixed++; else skipped++; // قراءة المحتوى بأمان let content; try { content = await entry.async('uint8array'); } catch (e2) { console.warn(`⚠️ Skip corrupted file: ${path}`); errors++; continue; } // حفظ مع تاريخ جديد outZip.file(path, content, { date: now, binary: true, createFolders: true }); } catch (fileError) { console.warn(`⚠️ Error processing ${path}:`, fileError.message); errors++; } } setZipProgress(88, 'Finalizing ZIP...'); await sleep(400); // إنشاء الـ ZIP النهائي zipState.fixedBlob = await outZip.generateAsync({ type: 'blob', compression: 'DEFLATE', compressionOptions: { level: 5 }, platform: 'UNIX' }, (update) => { const pct = 88 + Math.floor(update.percent * 0.12); setZipProgress(pct, `Compressing... ${Math.floor(update.percent)}%`); }); setZipProgress(100, '✅ FIXED!'); await sleep(800); // عرض النتائج const statsHtml = `
Files${entries.length}
Fixed${fixed}
OK${skipped}
${errors ? `
Errors${errors}
` : ''}
Output${formatSize(zipState.fixedBlob.size)}
`; document.getElementById('zipStatsEl').innerHTML = statsHtml; document.getElementById('zipStatsEl').classList.add('show'); document.getElementById('zipSuccessBadge').innerHTML = `✅ Success! Fixed ${fixed} files.
${errors ? `Skipped ${errors} corrupted files.` : ''}
Ready for Windows ✓`; document.getElementById('zipSuccessBadge').classList.add('show'); zipDownloadBtn.style.display = 'flex'; zipDownloadBtn.textContent = `⬇ ${zipState.outName}`; console.log('✅ ZIPFix complete!'); } catch (error) { console.error('❌ Final error:', error); setZipProgress(0, ''); let errorMsg = 'Unknown error'; if (error.message.includes('uncompressed')) { errorMsg = 'Invalid ZIP structure. Try another file.'; } else if (error.message.includes('CRC')) { errorMsg = 'ZIP corrupted. File may be incomplete.'; } else { errorMsg = error.message; } document.getElementById('zipErrorBadge').textContent = `❌ ${errorMsg}`; document.getElementById('zipErrorBadge').classList.add('show'); } fixBtn.disabled = false; fixBtn.textContent = '⚡ FIX AGAIN'; });
